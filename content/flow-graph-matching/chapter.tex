\section{Modelling with cuts and matchings}
\begin{itemize}[leftmargin=*,noitemsep]
	\item \textbf{Hall:} bipartite $(L\cup R,E)$ has a matching
	saturating $L$ iff $|N(S)|\ge|S|$ for all $S\subseteq L$.
	\item \textbf{K\"onig:} in bipartite graphs, max matching $=$ min
	vertex cover; max independent set $= |V| -$ max matching.
	\item \textbf{Project selection / max closure:} node weights $w_v$.
	$S\to v$ cap $w_v$ if $w_v>0$, $v\to T$ cap $-w_v$ if $w_v<0$; for
	each dependency ``$u$ needs $v$'' (edge $u\to v$) add $u\to v$ cap
	$\infty$. Best $=\sum_{w_v>0}w_v-\text{mincut}$; chosen set is the
	$S$ side. Projects needing machines: project weight $p_i$, machine
	weight $-q_j$. Selecting an edge needs both endpoints: treat edges
	as projects depending on 2 nodes.
	\item \textbf{Chains of choices:} to pick a value $h\in[0,H]$ per
	item, make nodes $(i,h)$ with $(i,h)\to(i,h+1)$ cap $\infty$
	(monotone), then cut costs become capacities on the chain.
	\item \textbf{Min path cover in DAG (vertex-disjoint):} split $v$
	into $v_L,v_R$; edge $u\to v$ becomes $u_L - v_R$. Answer $=
	|V| - $ max matching. Not vertex-disjoint: use the transitive
	closure first.
	\item \textbf{Max weight independent set, bipartite, $w\ge0$:}
	$S\to l$ cap $w_l$, $r\to T$ cap $w_r$, $l\to r$ cap $\infty$.
	Answer $=\sum w-\text{mincut}$.
\end{itemize}

\section{Flow with lower bounds $[L,R]$}
Edge $u\to v$ with bounds $[L,R]$: add $u\to v$ cap $R-L$,
$u\to T'$ cap $L$, $S'\to v$ cap $L$ ($S',T'$ new). Add $T\to S$
cap $\infty$. Feasible iff \texttt{maxFlow}$(S',T')=\sum L$. Max
flow: on the \emph{same} DinicFlow (keep $T\to S$) the value of
\texttt{maxFlow}$(S,T,$\texttt{false}$)$ is the answer. Min flow:
$f$ on $T\to S$ (edge $i$), remove it (zero \texttt{capa} and
\texttt{flow} of $i$, $i\oplus1$), answer
$f-$\texttt{maxFlow}$(T,S,$\texttt{false}$)$.
Assumes no edges into $S$ or out of $T$. Circulation (no
$S,T$): only the feasibility step.
